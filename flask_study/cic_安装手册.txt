CREATE TABLE `db_cic`.`t_user` (
	`id` bigint(20) NOT NULL AUTO_INCREMENT COMMENT '自增id',
	`uid` varchar(32) CHARACTER SET utf8 NOT NULL COMMENT '用户id.',
	`name` varchar(32) NOT NULL COMMENT '用户名',
	`password` varchar(128) NOT NULL COMMENT '密码',
	`mobile` int(11) NOT NULL COMMENT '手机号',
	`email` varchar(128) COMMENT '邮箱',
	`avatar_path` varchar(128) COMMENT '头像地址',
	`avatar_time` int(10) COMMENT '头像更新时间',
	`nick_name` varchar(32),
	`male` tinyint(1) UNSIGNED DEFAULT 0 COMMENT '性别。0：男；1：女',
	PRIMARY KEY (`id`),
	INDEX `mobile` (`mobile`) comment '',
	UNIQUE `uid` (`uid`) comment ''
) COMMENT='';
